  const logs = this.globalData.CheckinLogs || [];
  const habitIdStr = String(habitId);
  const date = dateStr || new Date().toISOString().split('T')[0];
  
  // 检查是否已存在今日记录

    log.habitId === habitIdStr && log.date === date
  );
  
  if (existingIndex === -1) {
    logs.push({
      logId: `L_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`,
      habitId: habitIdStr,
      date: date,
      timestamp: new Date(date).getTime()
    });
    this.saveCheckinLogs(logs);
    return true;
  }
  return false;
}

exports.main = async (event, context) => {
  const wxContext = cloud.getWXContext();
  const openid = wxContext.OPENID;
  const { habit_id } = event;
  
  const todayStr = formatDate(new Date());
  const habitIdStr = String(habit_id);

  // 1. 检查今日是否已打卡
  const existingLog = await db.collection('checkin_logs').where({
    _openid: openid,
    habit_id: habitIdStr,
    checkin_date: todayStr
  }).get();

  if (existingLog.data && existingLog.data.length > 0) {
    return { success: false, message: '今日已打卡' };
  }

  // 2. 验证习惯策略存在
  const strategyRes = await db.collection('user_strategies').where({
    _openid: openid,
    habit_id: habitIdStr
  }).get();

  if (!strategyRes.data || strategyRes.data.length === 0) {
    return { success: false, message: '未找到该习惯的策略' };
  }
  
  // 3. 写入打卡记录
  await db.collection('checkin_logs').add({
    data: {
      _openid: openid,
      habit_id: habitIdStr,
      checkin_date: todayStr,
      created_at: new Date()
    }
  });
  
  return { success: true, message: '打卡成功' };
};

## 五、核心计算逻辑

### 5.1 连续打卡天数计算

```javascript
// 计算习惯的连续打卡天数
calculateStreak(habitId) {
  const logs = this.getLogsByHabitId(habitId);
  if (logs.length === 0) return 0;
  
  // 按日期排序（降序）
  const sortedDates = logs
    .map(log => log.date)
    .sort((a, b) => new Date(b) - new Date(a));
  
  const today = new Date().toISOString().split('T')[0];
  const yesterday = new Date(Date.now() - 86400000)
    .toISOString().split('T')[0];
  
  // 确定连续性的起点（今天或昨天）
  let checkDate = sortedDates.includes(today) 
    ? today 
    : yesterday;
  
  if (!sortedDates.includes(checkDate)) {
    return 0;
  }
  
  // 向前推算连续天数
  let streak = 0;
  for (let i = 0; i < sortedDates.length; i++) {
    const currentDate = new Date(checkDate);
    currentDate.setDate(currentDate.getDate() - i);
    const dateStr = currentDate.toISOString().split('T')[0];
    
    if (sortedDates.includes(dateStr)) {
      streak++;
    } else {
      break;
    }
  }
  
  return streak;
}
```

### 5.2 打卡完成率计算（策略感知版）

```javascript
// 计算打卡完成率（根据习惯策略计算应打卡天数）
calculateStatsWithStrategy(habits, checkinLogs, weekStart, weekEnd) {
  let totalExpected = 0;  // 应打卡总次数
  let totalActual = 0;    // 实际打卡总次数

  habits.forEach(habit => {
    // 计算本周该习惯应打卡天数
    let expectedDays = 0;
    let actualDays = 0;

    for (let d = new Date(weekStart); d <= weekEnd; d.setDate(d.getDate() + 1)) {
      const shouldShow = this.shouldShowHabitOnDate(habit, new Date(d));
      if (shouldShow) {
        expectedDays++;
        const dateStr = this.formatDateKey(d);
        const hasCheckin = checkinLogs.some(
          log => log.habitId === habit.habitId && log.date === dateStr
        );
        if (hasCheckin) {
          actualDays++;
        }
      }
    }

    totalExpected += expectedDays;
    totalActual += actualDays;
  });

  const completionRate = totalExpected > 0 
    ? Math.round((totalActual / totalExpected) * 100) 
    : 0;

  return {
    totalExpected,
    totalActual,
    completionRate,
    habitsCount: habits.length
  };
}

// 旧版简单计算（已弃用）
calculateCheckinRateLegacy(checkinLogs, habits, startDate, endDate) {
  const totalDays = Math.round(
    (endDate - startDate) / (1000 * 60 * 60 * 24)
  ) + 1;
  
  // 应打卡次数 = 习惯数 × 天数（不考虑策略）
  const expectedCount = habits.length * totalDays;
  
  // 实际打卡次数
  const actualCount = checkinLogs.length;
  
  // 打卡率
  return expectedCount > 0 
    ? Math.round((actualCount / expectedCount) * 100) 
    : 0;
}
```

### 5.3 应打卡天数计算（calculateDueCount）

```javascript
// 计算从 startDate 到 endDate 之间的应打卡天数
// planStartDate: 计划开始时间（锚点，锚点之前的日期不计入）
function calculateDueCount(startDate, endDate, freqType, freqRules, planStartDate) {
  if (!startDate || !endDate || !planStartDate) {
    return 0;
  }

  const start = parseDate(startDate);
  const end = parseDate(endDate);
  const planStart = parseDate(planStartDate);

  if (end < planStart) {
    return 0;
  }

  const effectiveStart = start < planStart ? planStart : start;

  if (effectiveStart > end) {
    return 0;
  }

  const diffDays = Math.floor((end - effectiveStart) / (1000 * 60 * 60 * 24));

  // 每日策略：从计划开始日起，每一天都是应打卡日
  if (freqType === 'daily') {
    return diffDays + 1;
  }

  // 间隔策略：计划开始日是第1次，之后每隔 N 天打卡一次
  if (freqType === 'interval') {
    const intervalDays = freqRules || 1;
    if (diffDays < 0) {
      return 0;
    }
    return Math.floor(diffDays / intervalDays) + 1;
  }

  // 每周策略：从计划开始日起，每周重复用户选定的星期几
  if (freqType === 'weekly') {
    const targetDays = Array.isArray(freqRules) ? freqRules : [];
    if (targetDays.length === 0) {
      return diffDays + 1;
    }

    const totalWeeks = Math.floor(diffDays / 7);
    const fullWeekCount = totalWeeks * targetDays.length;

    const remainingDays = diffDays % 7;
    let remainingCount = 0;

    for (let i = 0; i <= remainingDays; i++) {
      const checkDate = new Date(effectiveStart);
      checkDate.setDate(effectiveStart.getDate() + i);
      const dayOfWeek = checkDate.getDay();
      const normalizedDay = dayOfWeek === 0 ? 7 : dayOfWeek; // 周日归一化为7
      if (targetDays.includes(normalizedDay)) {
        remainingCount++;
      }
    }

    return fullWeekCount + remainingCount;
  }

  return diffDays + 1;
}
```

**计算示例（基于 2026-05-05）：**

| 策略类型 | 设置 | 计算过程 | 应打卡天数 |
|---------|------|---------|-----------|
| 每日 | plan_start=5月1日 | diff=4, 4+1=5 | 5次 |
| 间隔3天 | plan_start=5月1日, interval=3 | diff=4, floor(4/3)+1=2 | 2次 |
| 每周二五 | plan_start=4月27日(周一), target=[2,5] | 1周×2=2, 剩余1天含1个目标日, 2+1=3 | 3次 |

### 5.4 策略变更与分段计算

当用户修改策略时，系统会保存策略版本历史，确保历史统计准确。

```javascript
// 策略版本记录格式
const strategyVersion = {
  habit_id: '1',
  freq_type: 'interval',      // 'daily' | 'weekly' | 'interval'
  freq_rules: 3,               // 间隔天数或每周目标日数组
  start_date: '2026-05-01',   // 该版本生效开始日期
  end_date: '2026-05-04'       // 该版本失效日期，null表示当前生效
};

// 分段计算逻辑
function calculateDueCountWithVersions(startDate, endDate, strategy, habitVersions) {
  const planStartDate = strategy.plan_start_date || strategy.created_at;
  
  if (!habitVersions || habitVersions.length === 0) {
    // 无版本变更，直接计算
    return calculateDueCount(
      startDate, endDate,
      strategy.freq_type, strategy.freq_rules,
      planStartDate
    );
  }

  let totalDue = 0;
  let segmentStart = startDate;

  // 按时间顺序遍历所有版本
  habitVersions.sort((a, b) => new Date(a.start_date) - new Date(b.start_date));

  habitVersions.forEach(version => {
    if (!version.start_date) return;

    const versionStart = version.start_date;
    const versionEnd = version.end_date || endDate;

    // 跳过已过期的版本段
    if (parseDate(versionEnd) < parseDate(planStartDate)) {
      segmentStart = versionEnd;
      return;
    }

    // 确定该版本的生效区间
    const segStart = parseDate(segmentStart) < parseDate(versionStart)
      ? versionStart
      : segmentStart;
    const segEnd = parseDate(versionEnd) < parseDate(endDate)
      ? versionEnd
      : endDate;

    // 计算该段内的应打卡天数
    if (parseDate(segStart) <= parseDate(segEnd)) {
      const dueCount = calculateDueCount(
        segStart, segEnd,
        version.freq_type, version.freq_rules,
        planStartDate
      );
      totalDue += dueCount;
    }

    segmentStart = versionEnd;
  });

  // 计算最后一个版本段到结束日期
  if (parseDate(segmentStart) <= parseDate(endDate)) {
    const dueCount = calculateDueCount(
      segmentStart, endDate,
      strategy.freq_type, strategy.freq_rules,
      planStartDate
    );
    totalDue += dueCount;
  }

  return totalDue;
}
```

### 5.5 最长连日计算（基于应打卡日）

最长连日必须基于"应打卡日"计算，而非自然日。如果某天是应打卡日但用户没打，连续中断。

```javascript
// 判断某天是否是应打卡日
function isExpectedCheckinDay(date, planStart, freqType, freqRules) {
  if (date < planStart) {
    return false;
  }

  const diffDays = Math.floor((date - planStart) / (1000 * 60 * 60 * 24));

  if (freqType === 'daily') {
    return true;
  }

  if (freqType === 'interval') {
    const intervalDays = freqRules || 1;
    return diffDays >= 0 && diffDays % intervalDays === 0;
  }

  if (freqType === 'weekly') {
    const targetDays = Array.isArray(freqRules) ? freqRules : [];
    if (targetDays.length === 0) {
      return true;
    }
    const dayOfWeek = date.getDay();
    const normalizedDay = dayOfWeek === 0 ? 7 : dayOfWeek;
    return targetDays.includes(normalizedDay);
  }

  return true;
}

// 计算最长连续应打卡日都完成打卡的天数
function calculateMaxStreak(checkinDates, planStartDate, freqType, freqRules) {
  const dateSet = new Set(checkinDates);
  const planStart = parseDate(planStartDate);
  const today = new Date();
  today.setHours(0, 0, 0, 0);

  let maxStreak = 0;
  let currentStreak = 0;
  let expectedDate = new Date(planStart);

  while (expectedDate <= today) {
    const dateStr = formatDate(expectedDate);
    const isDue = isExpectedCheckinDay(expectedDate, planStart, freqType, freqRules);

    if (isDue) {
      if (dateSet.has(dateStr)) {
        currentStreak++;
      } else {
        maxStreak = Math.max(maxStreak, currentStreak);
        currentStreak = 0;
      }
    }

    expectedDate.setDate(expectedDate.getDate() + 1);
  }

  return Math.max(maxStreak, currentStreak);
}
```

  const { habitId } = e.currentTarget.dataset;

  const isChecked = habit.isChecked;

  const taskList = this.data.taskList.map(item => {

        ...item,
        isChecked: !isChecked,
        streak: !isChecked 
          ? (item.streak || 0) + 1 
          : Math.max(0, (item.streak || 0) - 1)
      };
    }
    return item;
  });

  // 2. 更新进度
  const checkedCount = taskList.filter(item => item.isChecked).length;
  const progressPercent = Math.round((checkedCount / taskList.length) * 100);
  this.setData({ taskList, checkedCount, progressPercent });

  // 3. 更新全局数据
  if (!isChecked) {
    app.addCheckinLog(habitId, today);
  } else {
    app.removeCheckinLog(habitId, today);
  }

  // 4. 同步到云端
  wx.cloud.callFunction({
    name: !isChecked ? 'doCheckin' : 'undoCheckin',
    data: { habit_id: habitId }
  });
}

  // 计算连续打卡天数
  calculateStreak(habitId) {
    const logs = this.getLogsByHabitId(habitId);
    if (logs.length === 0) return 0;
    
    const sortedDates = logs
      .map(log => log.date)
      .sort((a, b) => new Date(b) - new Date(a));
    
    const today = new Date().toISOString().split('T')[0];
    const yesterday = new Date(Date.now() - 86400000)
      .toISOString().split('T')[0];
    
    let checkDate = sortedDates.includes(today) ? today : yesterday;
    if (!sortedDates.includes(checkDate)) return 0;
    
    let streak = 0;
    for (let i = 0; i < sortedDates.length; i++) {
      const currentDate = new Date(checkDate);
      currentDate.setDate(currentDate.getDate() - i);
      const dateStr = currentDate.toISOString().split('T')[0];
      
      if (sortedDates.includes(dateStr)) {
        streak++;
      } else {
        break;
      }
    }
    
    return streak;
  },

  // 修改习惯策略

    const habits = this.globalData.MyHabits || [];

        ...habits[index],
        ...newStrategy,
        updatedAt: new Date().toISOString()
      };

    return false;

### 11.2 删除习惯保留数据（完整实现）

删除习惯时保留历史打卡记录，确保观心页面能显示已删除习惯的历史数据：

```javascript
// 删除习惯（保留打卡数据）
removeStrategy(habit) {
  // 1. 保存习惯信息到 AllHabitsInfo
  app.saveDeletedHabitInfo(habit);
  
  // 2. 从MyHabits中移除
  app.removeHabit(habit.habitId);
  
  // 3. 保留CheckinLogs中的记录
  // 这样观心页面仍能看到历史打卡数据
}
```

**数据保留机制说明：**

1. **AllHabitsInfo 存储**：删除习惯时，将习惯的完整信息保存到本地存储 `AllHabitsInfo` 中，键为 habitId
2. **CheckinLogs 保留**：不删除 `CheckinLogs` 中的打卡记录，保留所有历史数据
3. **观心页面合并**：加载观心页面时，调用 `mergeWithDeletedHabits()` 方法，将当前习惯和已删除但有打卡记录的习惯合并
4. **图标主题恢复**：通过 `iconMap.getIconConfig()` 恢复已删除习惯的原始图标和主题色
5. **打卡完成率计算**：已删除习惯的打卡数据参与完成率计算，确保统计数据准确
